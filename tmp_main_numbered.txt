 235:     // Token management
 236:     loadTokensFromStorage() {
 237:         const stored = localStorage.getItem(this.tokenStorageKey);
 238:         if (stored) {
 239:             const data = JSON.parse(stored);
 240:             const today = new Date().toDateString();
 241:             if (data.date === today) {
 242:                 this.usedTokens = data.usedTokens || 0;
 243:             } else {
 244:                 this.usedTokens = 0;
 245:                 this.saveTokensToStorage();
 246:             }
 247:         }
 248:         this.updateTokenDisplay();
 249:     }
 250: 
 251:     saveTokensToStorage() {
 252:         const data = {
 253:             usedTokens: this.usedTokens,
 254:             date: new Date().toDateString()
 255:         };
 256:         localStorage.setItem(this.tokenStorageKey, JSON.stringify(data));
 257:     }
 258: 
 259:     hasTokensAvailable() {
 260:         return this.usedTokens < this.maxTokens;
 261:     }
 262: 
 263:     consumeToken() {
 264:         if (this.hasTokensAvailable()) {
 265:             this.usedTokens++;
 266:             this.saveTokensToStorage();
 267:             this.updateTokenDisplay();
 268:             return true;
 269:         }
 270:         return false;
 271:     }
 272: 
 273:     getRemainingTokens() {
 274:         return Math.max(0, this.maxTokens - this.usedTokens);
 275:     }

 277:     updateTokenDisplay() {
